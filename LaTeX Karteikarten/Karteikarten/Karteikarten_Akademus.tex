\begin{frage}
	Nullstellen einer Funktion?
	
\end{frage}
\begin{antwort}
	Ansatz: Funktion mit 0 gleichsetzen
	\[
	f(x)=0
	\]
\end{antwort}
\begin{frage}
	Nullstellen: Lineare Funktion
		
	z.B.: $f(x)=4x-12$
		
\end{frage}
\begin{antwort}
	Einfach umstellen:
	\[
	4x-12=0
	\]
	\[
	4x=12
	\]
	\[
	x=3
	\]
\end{antwort}
\begin{frage}
	Nullstellen: Quadratische Funktion
	\[
		f(x)=2x^2+6x-8
	\]
\end{frage}
\begin{antwort}
		\[
		2x^2+6x-8=0
		\]
	"`Mitternachtsformel"'
	\[
		x_{1,2}=\frac{-b\pm\sqrt{b^2-4ac}}{2a}
	\]
	\[
	x_{1,2}=\frac{-6\pm\sqrt{6^2-4\cdot2\cdot(-8)}}{2\cdot2 }
	\]
	\[
	x_{1}=1,\;x_{2}=-4
	\]

\end{antwort}
\begin{frage}
	Nullstellen: Funktion 3. Grades
\end{frage}
\begin{antwort}
	\normalsize\begin{enumerate}
		\item Schritt: $x$ Ausklammern
		\item Schritt: Mitternachtsformel in der Klammer anwenden
	\end{enumerate}
	\normalsize\begin{align*}
		2x^3+2x^2-24x&=0\\
		x\cdot(2x^2+2x-24)&=0\\
		=>	x_{1}&= 0\\
		x_{2,3}&= ...Mitternachtsformel\\
		=>	x_{2}=3,\;x_{3}&=-4\\
	\end{align*}	
\end{antwort}
\begin{frage}
	Nullstellen: Funktion 3. Grades
	\[
	f(x)=3x^3+81
	\]
\end{frage}
\begin{antwort}
	Auflösen:
		\[3x^3+81=0\]
		\[3x^3=-81\]
		\[x^3=-27\]
		\[x=\sqrt[3]{-27}=-9\]
		\normalsize
		(Die Wurzel einer negativen Zahl ist hier möglich, da es eine ungerade Wurzel ist)
\end{antwort}

\begin{frage}
	Symmetrie einer Funktion untersuchen
\end{frage}
\begin{antwort}
	\begin{enumerate}
		\item Setze $-x$ in die Funktion $f$ ein: $f(-x)$
		\item Vereinfache die Funktion
		\item Falls $f(-x)=f(x)$ dann liegt Achsensymmetrie zur y-Achse vor. Falls $f(-x)=-f(x)$ dann liegt Punktsymmetrie zum Ursprung vor.
	\end{enumerate}
\end{antwort}

\begin{frage}
	Waagrechte Asymptoten (beliebige Funktion)
\end{frage}
\begin{antwort}

		\item Berechne $\lim\limits_{x \rightarrow \infty}{f(x)}$ und $\lim\limits_{x \rightarrow -\infty}{f(x)}$	
		
		\item Wenn eine Zahl rauskommt, dann lautet die waagrechte Asymptote y = \textbf{diese Zahl}

	\end{enumerate}

	\end{align*}
\end{antwort}

\begin{frage}
	Verschiebung und Streckung von Funktionen  \bigskip
	
	$f(x)=a \cdot f(b(x+c))+d$
\end{frage}
\begin{antwort}
	\large
	$f(x)=a \cdot f(b(x+c))+d$
	\begin{itemize}

		\item a: Streckung in y-Richtung um den Faktor a
		\item b: Streckung in x-Richtung um den Faktor $\mathbf{\frac{1}{b}}$
		\item c: Verschiebung um \textbf{-c} in x-Richtung
		\item d: Verschiebung um d in y-Richtung
							
	\end{itemize}
	\textbf{Beachte:} Gilt für alle Funktionen.\\
	 Zuerst Streckungen dann Verschiebungen!
\end{antwort}

\begin{frage}
	Ableitregel für Polynome
	
	z.B.: $f(x)=3x^4+2x^3-7$
	
\end{frage}
\begin{antwort}
	Ableitregel Polynome:
	\[
	f(x) = n \cdot x^m
	\]
	\[	
	f'(x)= nm \cdot x^{m-1}
	\]
	Im Beispiel:
	\[
	f'(x)=12x^3+6x^2
	\]
\end{antwort}
\begin{frage}
	Produktregel
	
	z.B.: $f(x)=sin(x)\cdot2x^3$
	
\end{frage}
\begin{antwort}
	Produktregel:
	\[
	f(x) = u(x)\cdot v(x)
	\]
	\[
	f'(x)=u'(x)\cdot v(x)+u(x)\cdot v'(x)
	\]
	Im Beispiel:
	\[
	f'(x)=cos(x)\cdot 2x^3 + sin(x)\cdot 6x^2
	\]
\end{antwort}
\begin{frage}
	Quotientenregel
	
	z.B.: $f(x)=\frac{sin(x)}{3x^2}$
	
\end{frage}
\begin{antwort}
	Quotientenregel:
	\normalsize
	\[
	\frac{NAZ-ZAN}{N^2}=\frac{Nenner \cdot AbleitungZ{"a}hler-Z{"a}hler \cdot AbleitungNenner}{Nenner^2}
	\]
	\Large
	Im Beispiel:
	\[
	f'(x)=\frac{3x^2 \cdot cos(x) - sin(x) \cdot (6x)}{(3x^2)^2}
	\]
\end{antwort}
\begin{frage}
	Kettenregel
	
	z.B.: $f(x)={sin(3x^2)}$
	
\end{frage}
\begin{antwort}
	Kettenregel:
	\[
	(f(g(x)))'=f'(g(x))\cdot g'(x)
	\]
	Im Beispiel:
	\[
	f(x)=sin(x);~ g(x)=3x^2\]
	\[f'(x)=cos(3x^2)\cdot 6x
	\]
\end{antwort}

\begin{frage}
	Fläche zwischen zwei Funktionen $f(x)$ und $g(x)$
\end{frage}
\begin{antwort}
		\large
		\begin{enumerate} 
			\item Schnittstellen berechnen: $f(x)=g(x)$
			
			Man erhält die Integrationsgrenzen $x_{klein}$ und $x_{gro{"s}}$.
			\item Fläche berechnen: 
				$A=\abs{\displaystyle\int\limits_{x_{klein}}^{x_{gro{"s}}} \! (f(x)-g(x)) \, \mathrm{d}x}$
			
			\textbf{Beachte:} \Large Wenn 3 Schnittstellen statt 2 existieren, muss das Integral aufgeteilt werden:
			
				$\abs{\displaystyle\int\limits_{x_{klein}}^{x_{mittel}} \! (f(x)-g(x)) \, \mathrm{d}x}+\abs{\displaystyle\int\limits_{x_{mittel}}^{x_{gro{"s}}} \! (f(x)-g(x)) \, \mathrm{d}x}$
		\end{enumerate} 
\end{antwort}
